% Generic entity registry and rendering helpers.
\newcounter{characterCount}
\newcounter{spaceCount}
\newcounter{clueCount}
\newcounter{mysteryCount}
\newif\ifMMRegistrationPass
\MMRegistrationPassfalse

\makeatletter
% Character display names live in config/characters.tex so they can be kept
% separate from the player sheets and other game content.
\newcommand{\MMCharacterName}[2]{%
  \@ifundefined{mm@configured@character@#1}{%
    \expandafter\gdef\csname mm@configured@character@#1\endcsname{#2}%
  }{%
    \PackageError{murder-mystery}{Duplicate character identifier `#1'}%
      {Each character identifier must be named only once in config/characters.tex.}%
  }%
}

% A configured display name overrides the sheet title; the title is the
% fallback when an ID has no entry in config/characters.tex.
\newcommand{\MMConfiguredCharacterName}[2]{%
  \@ifundefined{mm@configured@character@#1}{%
    #2%
  }{%
    \csname mm@configured@character@#1\endcsname%
  }%
}

\newcommand{\MMDeclareNamedEntity}[4]{%
  \@ifundefined{mm@#1@#2}{%
    \expandafter\gdef\csname mm@#1@#2\endcsname{#3}%
    \stepcounter{#4}%
  }{%
    \PackageError{murder-mystery}{Duplicate #1 identifier `#2'}%
      {Each entity identifier must be declared only once in its own content file.}%
  }%
}

\newcommand{\MMNamedEntity}[2]{%
  \@ifundefined{mm@#1@#2}{%
    \PackageError{murder-mystery}{Unknown #1 identifier `#2'}%
      {Declare this identifier in its entity file and add that file to the matching index.tex.}%
    \textbf{[unknown #1: #2]}%
  }{%
    \csname mm@#1@#2\endcsname%
  }%
}

\newcommand{\MMEntityReference}[3]{%
  \@ifundefined{mm@#1@#2}{%
    \MMNamedEntity{#1}{#2}%
  }{%
    \hyperref[#1:#2]{#3{\MMNamedEntity{#1}{#2}}}%
  }%
}
\makeatother

% Character names can be configured in config/characters.tex. During the
% silent registration pass each sheet records its ID and display name, falling
% back to its title when no configured name exists; during normal rendering it
% typesets the body.
\newcommand{\CharacterSheet}[3]{%
  \ifMMRegistrationPass
    \MMDeclareNamedEntity{character}{#1}{\MMConfiguredCharacterName{#1}{#2}}{characterCount}%
  \else
    \subsubsection{#2: \CharacterReferenceStyle{\CharacterName{#1}}}
    \label{character:#1}%
    #3%
  \fi
}

\newcommand{\SpaceSheet}[3]{%
  \ifMMRegistrationPass
    \MMDeclareNamedEntity{space}{#1}{#2}{spaceCount}%
  \else
    \subsubsection{\SpaceReferenceStyle{\SpaceName{#1}}}
    \label{space:#1}%
    #3%
  \fi
}

\newcommand{\ClueSheet}[3]{%
  \ifMMRegistrationPass
    \MMDeclareNamedEntity{clue}{#1}{#2}{clueCount}%
  \else
    \subsubsection{\ClueReferenceStyle{\ClueName{#1}}}
    \label{clue:#1}%
    #3%
  \fi
}

% The mystery label is deliberately also its visible name and \Mystery{...}
% reference key.  This keeps a mystery declaration to a label and one body.
\newcommand{\MysterySheet}[2]{%
  \ifMMRegistrationPass
    \stepcounter{mysteryCount}%
  \else
    \subsubsection{#1}%
    \label{mystery:#1}%
    #2%
  \fi
}

% Declare that the first mystery must be solved before the second.  The
% generator reads these declarations to create Mermaid, SVG, and TikZ graphs;
% the declaration is intentionally silent in the rendered document.
\newcommand{\MysteryDependency}[2]{}

\newcommand{\RelationshipSummary}{%
  \subsection{Entity summary}
  There are \arabic{characterCount} characters.\par
  There are \arabic{clueCount} clues.\par
  There are \arabic{mysteryCount} mysteries.\par
}

% PDFLaTeX cannot embed SVG files directly. The generator writes a matching
% TikZ rendering, but the SVG remains the source-of-truth presence check so a
% normal game build simply omits the graph until relationship analysis exists.
\newcommand{\RelationshipGraph}{%
  \IfFileExists{generated/relationship_graph.svg}{%
    \IfFileExists{generated/relationship_graph.tex}{%
      \clearpage
      \subsection{Relationship graph}
      \input{generated/relationship_graph.tex}%
    }{%
      \PackageWarning{murder-mystery}{Relationship SVG exists but generated/relationship_graph.tex is missing. Run python compile.py}%
    }%
  }{}%
}

% As with the relationship graph, SVG is the existence check and the matching
% generated TikZ is the PDFLaTeX-native rendering.
\newcommand{\MysteryOrderGraph}{%
  \IfFileExists{generated/mystery_order_graph.svg}{%
    \IfFileExists{generated/mystery_order_graph.tex}{%
      \subsubsection{Mystery resolution order}%
      \input{generated/mystery_order_graph.tex}%
    }{%
      \PackageWarning{murder-mystery}{Mystery-order SVG exists but generated/mystery_order_graph.tex is missing. Run python compile.py}%
    }%
  }{}%
}
